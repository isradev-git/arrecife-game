extends SceneTree
func _initialize() -> void: call_deferred('check')
func check() -> void:
 var vp=SubViewport.new()
 vp.size=Vector2i(1600,1200)
 vp.render_target_update_mode=SubViewport.UPDATE_ALWAYS
 root.add_child(vp)
 var bg=ColorRect.new()
 bg.size=Vector2(1600,1200)
 bg.color=Color('#253040')
 bg.z_index=-100
 vp.add_child(bg)
 for profile in range(2):
  for i in range(4):
   var rig=load('res://sota_articulada.gd').new()
   vp.add_child(rig)
   rig.build(profile)
   rig.scale=Vector2.ONE*2
   rig.position=Vector2(200+i*400,550+profile*590)
   rig.pose(i/4.0)
   if '--sin-brazos' in OS.get_cmdline_user_args():
    for arm in rig.arms:
     arm.sleeve.hide()
     arm.hand.hide()
   var label=Label.new()
   label.text=('Derecha' if profile==0 else 'Izquierda')+' · '+str(i*8)
   label.position=Vector2(i*400+100,25+profile*590)
   vp.add_child(label)
 for n in range(5): await process_frame
 await RenderingServer.frame_post_draw
 var path=OS.get_cmdline_user_args()[0]
 assert(vp.get_texture().get_image().save_png(path)==OK)
 quit()
